cmake_minimum_required(VERSION 3.26)
project(LinkSpanPuppets LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(OOT_NATIVE_SDK "" CACHE FILEPATH "OotNativeSdk.cmake gerado pelo host")
set(HOST_EXECUTABLE "" CACHE FILEPATH "Exe exato para os três símbolos privados do desenho")
if(NOT EXISTS "${OOT_NATIVE_SDK}" OR NOT EXISTS "${HOST_EXECUTABLE}")
    message(FATAL_ERROR "Informe OOT_NATIVE_SDK e HOST_EXECUTABLE do mesmo build Release")
endif()
include("${OOT_NATIVE_SDK}")
string(LENGTH "${OOT_NATIVE_LAYOUT_ID}" layout_length)
if(NOT layout_length EQUAL 64 OR NOT OOT_NATIVE_LAYOUT_ID MATCHES "^[0-9a-f]+$")
    message(FATAL_ERROR "SDK sem layout id válido")
endif()

add_library(puppets SHARED provider.cpp ShipLuaPuppet.cpp)
target_include_directories(puppets PRIVATE ${OOT_NATIVE_INCLUDE_DIRS})
target_compile_definitions(puppets PRIVATE ${OOT_NATIVE_COMPILE_DEFINITIONS})
if(MSVC)
    target_compile_options(puppets PRIVATE /utf-8 /Zc:preprocessor)
    set_property(TARGET puppets PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set_target_properties(puppets PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/mod/provider")
foreach(config Debug Release RelWithDebInfo MinSizeRel)
    string(TOUPPER "${config}" upper)
    set_target_properties(puppets PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${upper} "${CMAKE_BINARY_DIR}/mod/provider")
endforeach()
file(SHA256 "${HOST_EXECUTABLE}" host_fingerprint)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS manifest.toml "${HOST_EXECUTABLE}")
file(READ manifest.toml manifest_text)
file(WRITE "${CMAKE_BINARY_DIR}/mod/manifest.toml"
    "${manifest_text}host_layout_id = \"${OOT_NATIVE_LAYOUT_ID}\"\nhost_fingerprints = [\"${host_fingerprint}\"]\n")
configure_file(main.lua "${CMAKE_BINARY_DIR}/mod/main.lua" COPYONLY)
configure_file(README.md "${CMAKE_BINARY_DIR}/mod/README.md" COPYONLY)

include(CTest)
if(BUILD_TESTING AND WIN32)
    add_executable(puppets_tests tests.cpp)
    target_include_directories(puppets_tests PRIVATE ${OOT_NATIVE_INCLUDE_DIRS})
    target_compile_definitions(puppets_tests PRIVATE ${OOT_NATIVE_COMPILE_DEFINITIONS})
    if(MSVC)
        target_compile_options(puppets_tests PRIVATE /utf-8 /Zc:preprocessor)
        set_property(TARGET puppets_tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
    add_dependencies(puppets_tests puppets)
    add_test(NAME puppets_contract COMMAND puppets_tests "$<TARGET_FILE:puppets>")
endif()
