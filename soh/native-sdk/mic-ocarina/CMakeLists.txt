cmake_minimum_required(VERSION 3.26)
project(MicOcarina LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(OOT_NATIVE_SDK "" CACHE FILEPATH "OotNativeSdk.cmake gerado pelo host")
if(NOT EXISTS "${OOT_NATIVE_SDK}")
    message(FATAL_ERROR "Informe -DOOT_NATIVE_SDK=<build-do-host>/native-sdk/Release/OotNativeSdk.cmake")
endif()
include("${OOT_NATIVE_SDK}")

find_package(SDL2 CONFIG REQUIRED)

add_library(mic_ocarina SHARED provider.cpp pitch_matcher.cpp package_assets.cpp)
add_executable(mic_ocarina_tests tests.cpp pitch_matcher.cpp)
add_executable(mic_ocarina_ui_tests remake_ui_tests.cpp pitch_matcher.cpp)
foreach(target mic_ocarina mic_ocarina_tests mic_ocarina_ui_tests)
    target_include_directories(${target} PRIVATE ${OOT_NATIVE_INCLUDE_DIRS} "${CMAKE_CURRENT_SOURCE_DIR}")
    target_compile_definitions(${target} PRIVATE ${OOT_NATIVE_COMPILE_DEFINITIONS})
    if(MSVC)
        target_compile_options(${target} PRIVATE /utf-8 /Zc:preprocessor)
        set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
endforeach()
target_compile_definitions(mic_ocarina PRIVATE SDL_MAIN_HANDLED)
target_link_libraries(mic_ocarina PRIVATE SDL2::SDL2-static)
target_compile_definitions(mic_ocarina_ui_tests PRIVATE SDL_MAIN_HANDLED)
target_link_libraries(mic_ocarina_ui_tests PRIVATE SDL2::SDL2-static)

set(provider_dir "${CMAKE_BINARY_DIR}/mod/provider")
set_target_properties(mic_ocarina PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${provider_dir}")
foreach(config Debug Release RelWithDebInfo MinSizeRel)
    string(TOUPPER "${config}" upper)
    set_target_properties(mic_ocarina PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${upper} "${provider_dir}")
endforeach()
configure_file(main.lua "${CMAKE_BINARY_DIR}/mod/main.lua" COPYONLY)
set(HOST_EXECUTABLE "" CACHE FILEPATH "Exact host executable for the ocarina renderer/input bindings")
if(NOT EXISTS "${HOST_EXECUTABLE}")
    message(FATAL_ERROR "Provide HOST_EXECUTABLE for the pinned ocarina bindings")
endif()
file(SHA256 "${HOST_EXECUTABLE}" host_fingerprint)
# O manifesto gerado é escrito no configure: sem o manifest.toml como dependência, um bump de versão não chegava ao pacote.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${HOST_EXECUTABLE}" manifest.toml)
file(READ manifest.toml manifest_text)
file(WRITE "${CMAKE_BINARY_DIR}/mod/manifest.toml" "${manifest_text}\nhost_fingerprints = [\"${host_fingerprint}\"]\n")
configure_file(README.md "${CMAKE_BINARY_DIR}/mod/README.md" COPYONLY)
file(COPY assets DESTINATION "${CMAKE_BINARY_DIR}/mod")

enable_testing()
add_test(NAME mic_ocarina_matcher COMMAND mic_ocarina_tests)
add_test(NAME mic_ocarina_remake_ui COMMAND mic_ocarina_ui_tests)
get_filename_component(sdk_directory "${OOT_NATIVE_SDK}" DIRECTORY)
get_filename_component(host_build "${sdk_directory}/../.." ABSOLUTE)
set(LUA_TEST_LIBRARY "${host_build}/ship-lua/Release/lua_static.lib" CACHE FILEPATH "Lua from the host, for HUD budget tests")
if(EXISTS "${LUA_TEST_LIBRARY}")
    add_executable(mic_ocarina_hud_tests hud_tests.cpp)
    target_include_directories(mic_ocarina_hud_tests PRIVATE ${OOT_NATIVE_INCLUDE_DIRS})
    if(MSVC)
        set_property(TARGET mic_ocarina_hud_tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
    target_link_libraries(mic_ocarina_hud_tests PRIVATE "${LUA_TEST_LIBRARY}")
    add_test(NAME mic_ocarina_hud_budget COMMAND mic_ocarina_hud_tests
        "${CMAKE_CURRENT_SOURCE_DIR}/hud_tests.lua" "${CMAKE_CURRENT_SOURCE_DIR}/main.lua")
endif()
