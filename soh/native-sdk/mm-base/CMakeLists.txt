cmake_minimum_required(VERSION 3.26)
project(MmBase LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(OOT_NATIVE_SDK "" CACHE FILEPATH "SDK exportado pelo host")
set(HOST_EXECUTABLE "" CACHE FILEPATH "Exe correspondente aos símbolos privados")
if(NOT EXISTS "${OOT_NATIVE_SDK}" OR NOT EXISTS "${HOST_EXECUTABLE}")
    message(FATAL_ERROR "Informe OOT_NATIVE_SDK e HOST_EXECUTABLE")
endif()
include("${OOT_NATIVE_SDK}")
if(NOT OOT_NATIVE_LAYOUT_ID MATCHES "^[0-9a-f]+$")
    message(FATAL_ERROR "SDK sem layout id")
endif()
file(GLOB_RECURSE audio_sources CONFIGURE_DEPENDS "mmaudio/*.cpp")
add_library(mm_base SHARED provider.cpp mount.cpp bindings.cpp ${audio_sources})
target_include_directories(mm_base PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" ${OOT_NATIVE_INCLUDE_DIRS})
set(mm_definitions ${OOT_NATIVE_COMPILE_DEFINITIONS})
list(REMOVE_ITEM mm_definitions SPDLOG_COMPILED_LIB SPDLOG_FMT_EXTERNAL)
target_compile_definitions(mm_base PRIVATE ${mm_definitions} FMT_HEADER_ONLY
    MM_BASE_LAYOUT_ID="${OOT_NATIVE_LAYOUT_ID}")
if(MSVC)
    target_compile_options(mm_base PRIVATE /utf-8 /Zc:preprocessor)
    set_property(TARGET mm_base PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
foreach(config Debug Release RelWithDebInfo MinSizeRel)
    string(TOUPPER "${config}" upper)
    set_target_properties(mm_base PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${upper} "${CMAKE_BINARY_DIR}/mod/provider")
endforeach()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${HOST_EXECUTABLE}" manifest.toml)
file(SHA256 "${HOST_EXECUTABLE}" fingerprint)
file(READ manifest.toml manifest_text)
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/mod")
configure_file(main.lua "${CMAKE_BINARY_DIR}/mod/main.lua" COPYONLY)
file(WRITE "${CMAKE_BINARY_DIR}/mod/manifest.toml"
    "${manifest_text}\nhost_layout_id = \"${OOT_NATIVE_LAYOUT_ID}\"\nhost_fingerprints = [\"${fingerprint}\"]\n")

include(CTest)
if(BUILD_TESTING AND WIN32)
    add_executable(mm_base_tests tests.cpp mount.cpp bindings.cpp mmaudio/MmSfxPlayer.cpp mmaudio/MmSoundFont.cpp)
    target_include_directories(mm_base_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" ${OOT_NATIVE_INCLUDE_DIRS})
    target_compile_definitions(mm_base_tests PRIVATE ${mm_definitions} FMT_HEADER_ONLY
        MM_BASE_LAYOUT_ID="${OOT_NATIVE_LAYOUT_ID}")
    if(MSVC)
        target_compile_options(mm_base_tests PRIVATE /utf-8 /Zc:preprocessor)
        set_property(TARGET mm_base_tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
    add_dependencies(mm_base_tests mm_base)
    add_test(NAME mm_base_shell COMMAND mm_base_tests "$<TARGET_FILE:mm_base>")
endif()
