cmake_minimum_required(VERSION 3.26)
project(DynamicMovementRemake LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(OOT_NATIVE_SDK "" CACHE FILEPATH "OotNativeSdk.cmake gerado pelo host")
set(JUMP_VELOCITY "7.0" CACHE STRING "Impulso implementado pelo mod")
if(NOT EXISTS "${OOT_NATIVE_SDK}")
    message(FATAL_ERROR "Informe -DOOT_NATIVE_SDK=<build-do-host>/native-sdk/Release/OotNativeSdk.cmake")
endif()
include("${OOT_NATIVE_SDK}")
add_library(dynamic_movement_remake SHARED provider.cpp package_assets.cpp)
target_include_directories(dynamic_movement_remake PRIVATE ${OOT_NATIVE_INCLUDE_DIRS})
target_compile_definitions(dynamic_movement_remake PRIVATE ${OOT_NATIVE_COMPILE_DEFINITIONS} JUMP_VELOCITY=${JUMP_VELOCITY})
if(MSVC)
    target_compile_options(dynamic_movement_remake PRIVATE /utf-8 /Zc:preprocessor)
endif()
# Nenhum link com shiplua ou soh: apenas headers e a tabela recebida no init.
set_target_properties(dynamic_movement_remake PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/mod/provider")
foreach(config Debug Release RelWithDebInfo MinSizeRel)
    string(TOUPPER "${config}" upper)
    set_target_properties(dynamic_movement_remake PROPERTIES
        RUNTIME_OUTPUT_DIRECTORY_${upper} "${CMAKE_BINARY_DIR}/mod/provider")
endforeach()
configure_file(main.lua "${CMAKE_BINARY_DIR}/mod/main.lua" COPYONLY)
get_filename_component(host_root "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(HOST_EXECUTABLE "${host_root}/x64/Release/soh.exe" CACHE FILEPATH "Exact host executable for the icon table")
if(NOT EXISTS "${HOST_EXECUTABLE}")
    message(FATAL_ERROR "Provide HOST_EXECUTABLE for the native icon-table binding")
endif()
file(SHA256 "${HOST_EXECUTABLE}" host_fingerprint)
# O manifesto gerado é escrito no configure: sem o manifest.toml como dependência, um bump de versão não chegava ao pacote.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${HOST_EXECUTABLE}" manifest.toml)
file(READ manifest.toml manifest_text)
file(WRITE "${CMAKE_BINARY_DIR}/mod/manifest.toml" "${manifest_text}\nhost_fingerprints = [\"${host_fingerprint}\"]\n")
# Texturas do mod no formato de recurso do libultraship; o provider monta a pasta no configure.
file(COPY assets DESTINATION "${CMAKE_BINARY_DIR}/mod")
# Licença (CC0 do trabalho do Link-Span) e atribuição vão no pacote.
configure_file(NOTICE.md "${CMAKE_BINARY_DIR}/mod/NOTICE.md" COPYONLY)
configure_file("${CMAKE_CURRENT_SOURCE_DIR}/../../../docs/licensing/LICENSE-PACKAGE.txt" "${CMAKE_BINARY_DIR}/mod/LICENSE" COPYONLY)
add_executable(dynamic_movement_icon_tests item_icons_tests.cpp)
enable_testing()
add_test(NAME dynamic_movement_icon_tests COMMAND dynamic_movement_icon_tests)
add_executable(dynamic_movement_camera_tests camera_continuity_tests.cpp)
target_include_directories(dynamic_movement_camera_tests PRIVATE ${OOT_NATIVE_INCLUDE_DIRS})
target_compile_definitions(dynamic_movement_camera_tests PRIVATE ${OOT_NATIVE_COMPILE_DEFINITIONS} JUMP_VELOCITY=${JUMP_VELOCITY})
if(MSVC)
    target_compile_options(dynamic_movement_camera_tests PRIVATE /utf-8 /Zc:preprocessor)
endif()
add_test(NAME dynamic_movement_camera_tests COMMAND dynamic_movement_camera_tests)
add_executable(dynamic_movement_menu_tests menu_input_tests.cpp)
target_include_directories(dynamic_movement_menu_tests PRIVATE ${OOT_NATIVE_INCLUDE_DIRS})
target_compile_definitions(dynamic_movement_menu_tests PRIVATE ${OOT_NATIVE_COMPILE_DEFINITIONS} JUMP_VELOCITY=${JUMP_VELOCITY})
if(MSVC)
    target_compile_options(dynamic_movement_menu_tests PRIVATE /utf-8 /Zc:preprocessor)
endif()
add_test(NAME dynamic_movement_menu_tests COMMAND dynamic_movement_menu_tests)
