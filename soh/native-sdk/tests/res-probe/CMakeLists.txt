cmake_minimum_required(VERSION 3.26)
project(ResourcesV4Probe LANGUAGES CXX)
set(OOT_NATIVE_SDK "" CACHE FILEPATH "OotNativeSdk.cmake do host F2g")
if(NOT EXISTS "${OOT_NATIVE_SDK}")
    message(FATAL_ERROR "Informe OOT_NATIVE_SDK da F2g; não usar headers do RC10/F2d")
endif()
include("${OOT_NATIVE_SDK}")
if(NOT OOT_NATIVE_LAYOUT_ID MATCHES "^[0-9a-f]+$")
    message(FATAL_ERROR "SDK sem layout id")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_library(res_probe SHARED provider.cpp)
target_compile_features(res_probe PRIVATE cxx_std_20)
target_include_directories(res_probe PRIVATE ${OOT_NATIVE_INCLUDE_DIRS})
set_target_properties(res_probe PROPERTIES PREFIX "")
foreach(config Debug Release RelWithDebInfo MinSizeRel)
    string(TOUPPER "${config}" upper)
    set_target_properties(res_probe PROPERTIES
        RUNTIME_OUTPUT_DIRECTORY_${upper} "${CMAKE_BINARY_DIR}/mod/provider"
        LIBRARY_OUTPUT_DIRECTORY_${upper} "${CMAKE_BINARY_DIR}/mod/provider")
endforeach()
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/mod")
configure_file(main.lua "${CMAKE_BINARY_DIR}/mod/main.lua" COPYONLY)
file(READ manifest.toml manifest_text)
file(WRITE "${CMAKE_BINARY_DIR}/mod/manifest.toml"
    "${manifest_text}host_layout_id = \"${OOT_NATIVE_LAYOUT_ID}\"\n")
add_custom_command(OUTPUT "${CMAKE_BINARY_DIR}/mod/res-probe.o2r"
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/make-fixture.py"
            "${CMAKE_BINARY_DIR}/mod/res-probe.o2r"
    DEPENDS make-fixture.py VERBATIM)
add_custom_target(res_probe_fixture ALL DEPENDS "${CMAKE_BINARY_DIR}/mod/res-probe.o2r")
option(RES_PROBE_OFFLINE_HARNESS "Compilar mock que testa a logica do probe, sem host/VFS real" OFF)
if(RES_PROBE_OFFLINE_HARNESS)
    enable_testing()
    add_executable(res_probe_offline offline_harness.cpp)
    target_compile_features(res_probe_offline PRIVATE cxx_std_20)
    target_include_directories(res_probe_offline PRIVATE ${OOT_NATIVE_INCLUDE_DIRS})
    add_test(NAME res_probe_offline COMMAND res_probe_offline)
endif()
