cmake_minimum_required(VERSION 3.26)
project(MmForms LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(OOT_NATIVE_SDK "" CACHE FILEPATH "SDK exportado do host")
set(HOST_EXECUTABLE "" CACHE FILEPATH "Exe para os símbolos privados")
set(MM_BASE_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../mm-base" CACHE PATH "Contrato da tarefa A")
if(NOT EXISTS "${OOT_NATIVE_SDK}" OR NOT EXISTS "${HOST_EXECUTABLE}" OR NOT EXISTS "${MM_BASE_INCLUDE_DIR}/mm_audio.h")
    message(FATAL_ERROR "Informe OOT_NATIVE_SDK, HOST_EXECUTABLE e MM_BASE_INCLUDE_DIR com mm_audio.h")
endif()
include("${OOT_NATIVE_SDK}")
set(MM_FORMS_MSVC_INCLUDE_PREFIX "" CACHE STRING "Correção local opcional do prefixo /showIncludes")
if(MSVC AND MM_FORMS_MSVC_INCLUDE_PREFIX)
    set(CMAKE_CL_SHOWINCLUDES_PREFIX "${MM_FORMS_MSVC_INCLUDE_PREFIX}")
endif()
add_library(mm_forms SHARED provider.cpp MmGoronForm.cpp game_bindings.cpp)
target_include_directories(mm_forms PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${MM_BASE_INCLUDE_DIR}" ${OOT_NATIVE_INCLUDE_DIRS})
set(forms_definitions ${OOT_NATIVE_COMPILE_DEFINITIONS})
list(REMOVE_ITEM forms_definitions SPDLOG_COMPILED_LIB SPDLOG_FMT_EXTERNAL)
target_compile_definitions(mm_forms PRIVATE ${forms_definitions} FMT_HEADER_ONLY MM_FORMS_LAYOUT_ID="${OOT_NATIVE_LAYOUT_ID}")
if(MSVC)
    target_compile_options(mm_forms PRIVATE /utf-8 /Zc:preprocessor)
    set_property(TARGET mm_forms PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
foreach(config Debug Release RelWithDebInfo MinSizeRel)
    string(TOUPPER "${config}" upper)
    set_target_properties(mm_forms PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${upper} "${CMAKE_BINARY_DIR}/mod/provider")
endforeach()
set_target_properties(mm_forms PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/mod/provider")
file(SHA256 "${HOST_EXECUTABLE}" fingerprint)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${HOST_EXECUTABLE}" manifest.toml)
file(READ manifest.toml manifest_text)
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/mod")
file(WRITE "${CMAKE_BINARY_DIR}/mod/manifest.toml"
    "${manifest_text}\nhost_layout_id = \"${OOT_NATIVE_LAYOUT_ID}\"\nhost_fingerprints = [\"${fingerprint}\"]\n")
configure_file(main.lua "${CMAKE_BINARY_DIR}/mod/main.lua" COPYONLY)
configure_file(README.md "${CMAKE_BINARY_DIR}/mod/README.md" COPYONLY)
configure_file(NOTICE.md "${CMAKE_BINARY_DIR}/mod/NOTICE.md" COPYONLY)
include(CTest)
if(BUILD_TESTING AND WIN32)
    add_executable(mm_forms_tests tests.cpp)
    target_include_directories(mm_forms_tests PRIVATE "${MM_BASE_INCLUDE_DIR}" ${OOT_NATIVE_INCLUDE_DIRS})
    target_compile_definitions(mm_forms_tests PRIVATE ${forms_definitions} MM_FORMS_LAYOUT_ID="${OOT_NATIVE_LAYOUT_ID}")
    if(MSVC)
        target_compile_options(mm_forms_tests PRIVATE /utf-8 /Zc:preprocessor)
    endif()
    add_dependencies(mm_forms_tests mm_forms)
    add_test(NAME mm_forms_shell COMMAND mm_forms_tests "$<TARGET_FILE:mm_forms>")
    get_filename_component(sdk_directory "${OOT_NATIVE_SDK}" DIRECTORY)
    get_filename_component(host_build "${sdk_directory}/../.." ABSOLUTE)
    set(LUA_TEST_LIBRARY "${host_build}/ship-lua/Release/lua_static.lib" CACHE FILEPATH "Lua só para o teste, fora da DLL")
    if(EXISTS "${LUA_TEST_LIBRARY}")
        add_executable(mm_forms_lua_tests lua_tests.cpp)
        target_include_directories(mm_forms_lua_tests PRIVATE ${OOT_NATIVE_INCLUDE_DIRS})
        target_link_libraries(mm_forms_lua_tests PRIVATE "${LUA_TEST_LIBRARY}")
        if(MSVC)
            set_property(TARGET mm_forms_lua_tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
        endif()
        add_test(NAME mm_forms_lua COMMAND mm_forms_lua_tests "${CMAKE_CURRENT_SOURCE_DIR}/lua_tests.lua"
                  "${CMAKE_CURRENT_SOURCE_DIR}/main.lua")
    endif()
endif()
