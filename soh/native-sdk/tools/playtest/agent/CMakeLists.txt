cmake_minimum_required(VERSION 3.26)
project(LinkSpanPlaytest LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(vfs_revision_material "")
foreach(source agent.cpp server.cpp server.h protocol.cpp protocol.h vfs.cpp vfs.h CMakeLists.txt)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${source}")
    file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${source}" source_sha)
    string(APPEND vfs_revision_material "${source}:${source_sha}\n")
endforeach()
string(SHA256 PLAYTEST_VFS_REVISION "${vfs_revision_material}")
set(OOT_NATIVE_SDK "" CACHE FILEPATH "SDK exportado pelo host")
set(HOST_EXECUTABLE "" CACHE FILEPATH "Exe do mesmo build")
if(NOT EXISTS "${OOT_NATIVE_SDK}" OR NOT EXISTS "${HOST_EXECUTABLE}")
    message(FATAL_ERROR "Informe OOT_NATIVE_SDK e HOST_EXECUTABLE")
endif()
include("${OOT_NATIVE_SDK}")
if(NOT WIN32 OR NOT OOT_NATIVE_LAYOUT_ID MATCHES "^[0-9a-f]+$")
    message(FATAL_ERROR "O agente exige Windows e SDK com layout id")
endif()
set(definitions ${OOT_NATIVE_COMPILE_DEFINITIONS})
list(REMOVE_ITEM definitions SPDLOG_COMPILED_LIB SPDLOG_FMT_EXTERNAL)
function(playtest_options target)
    target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" ${OOT_NATIVE_INCLUDE_DIRS})
    target_compile_definitions(${target} PRIVATE ${definitions} FMT_HEADER_ONLY NOMINMAX WIN32_LEAN_AND_MEAN
        PLAYTEST_LAYOUT_ID="${OOT_NATIVE_LAYOUT_ID}" PLAYTEST_VFS_REVISION="${PLAYTEST_VFS_REVISION}")
    if(MSVC)
        target_compile_options(${target} PRIVATE /utf-8 /Zc:preprocessor)
        set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
endfunction()
add_library(linkspan_playtest SHARED agent.cpp gameplay.cpp protocol.cpp queue.cpp server.cpp vfs.cpp)
playtest_options(linkspan_playtest)
target_link_libraries(linkspan_playtest PRIVATE ws2_32 bcrypt)
set_target_properties(linkspan_playtest PROPERTIES PREFIX "" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/mod/provider")
foreach(config Debug Release RelWithDebInfo MinSizeRel)
    string(TOUPPER "${config}" upper)
    set_target_properties(linkspan_playtest PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${upper} "${CMAKE_BINARY_DIR}/mod/provider")
endforeach()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS manifest.toml)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/manifest.toml" manifest_text)
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/mod")
file(WRITE "${CMAKE_BINARY_DIR}/mod/main.lua" "-- Ferramenta nativa; a entrada é registrada pelo provider.\n")
file(WRITE "${CMAKE_BINARY_DIR}/mod/manifest.toml" "${manifest_text}\nhost_layout_id = \"${OOT_NATIVE_LAYOUT_ID}\"\n")
include(CTest)
if(BUILD_TESTING)
    add_executable(playtest_agent_tests tests.cpp protocol.cpp queue.cpp vfs.cpp)
    playtest_options(playtest_agent_tests)
    target_link_libraries(playtest_agent_tests PRIVATE bcrypt)
    add_test(NAME playtest_agent_tests COMMAND playtest_agent_tests)
    add_executable(playtest_gameplay_tests gameplay_tests.cpp gameplay.cpp protocol.cpp queue.cpp server.cpp vfs.cpp)
    playtest_options(playtest_gameplay_tests)
    target_link_libraries(playtest_gameplay_tests PRIVATE ws2_32 bcrypt)
    add_test(NAME playtest_gameplay_tests COMMAND playtest_gameplay_tests)
endif()
