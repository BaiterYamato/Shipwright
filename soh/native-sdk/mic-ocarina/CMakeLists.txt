cmake_minimum_required(VERSION 3.26)
project(MicOcarina LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(OOT_NATIVE_SDK "" CACHE FILEPATH "OotNativeSdk.cmake gerado pelo host")
if(NOT EXISTS "${OOT_NATIVE_SDK}")
    message(FATAL_ERROR "Informe -DOOT_NATIVE_SDK=<build-do-host>/native-sdk/Release/OotNativeSdk.cmake")
endif()
include("${OOT_NATIVE_SDK}")

find_package(SDL2 CONFIG REQUIRED)

add_library(mic_ocarina SHARED provider.cpp pitch_matcher.cpp)
add_executable(mic_ocarina_tests tests.cpp pitch_matcher.cpp)
foreach(target mic_ocarina mic_ocarina_tests)
    target_include_directories(${target} PRIVATE ${OOT_NATIVE_INCLUDE_DIRS} "${CMAKE_CURRENT_SOURCE_DIR}")
    target_compile_definitions(${target} PRIVATE ${OOT_NATIVE_COMPILE_DEFINITIONS})
    if(MSVC)
        target_compile_options(${target} PRIVATE /utf-8 /Zc:preprocessor)
        set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
endforeach()
target_compile_definitions(mic_ocarina PRIVATE SDL_MAIN_HANDLED)
target_link_libraries(mic_ocarina PRIVATE SDL2::SDL2-static)

set(provider_dir "${CMAKE_BINARY_DIR}/mod/provider")
set_target_properties(mic_ocarina PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${provider_dir}")
foreach(config Debug Release RelWithDebInfo MinSizeRel)
    string(TOUPPER "${config}" upper)
    set_target_properties(mic_ocarina PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${upper} "${provider_dir}")
endforeach()
configure_file(main.lua "${CMAKE_BINARY_DIR}/mod/main.lua" COPYONLY)
configure_file(manifest.toml "${CMAKE_BINARY_DIR}/mod/manifest.toml" COPYONLY)
configure_file(README.md "${CMAKE_BINARY_DIR}/mod/README.md" COPYONLY)

enable_testing()
add_test(NAME mic_ocarina_matcher COMMAND mic_ocarina_tests)
