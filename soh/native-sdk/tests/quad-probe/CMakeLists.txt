cmake_minimum_required(VERSION 3.26)
project(PlayerQuadProbe LANGUAGES C)
set(OOT_NATIVE_SDK "" CACHE FILEPATH "OotNativeSdk.cmake do host F2f")
if(NOT EXISTS "${OOT_NATIVE_SDK}")
    message(FATAL_ERROR "Informe -DOOT_NATIVE_SDK=<build>/native-sdk/Release/OotNativeSdk.cmake")
endif()
include("${OOT_NATIVE_SDK}")
string(LENGTH "${OOT_NATIVE_LAYOUT_ID}" layout_length)
if(NOT OOT_NATIVE_LAYOUT_ID MATCHES "^[0-9a-f]+$" OR NOT layout_length EQUAL 64)
    message(FATAL_ERROR "SDK sem layout id SHA256 valido")
endif()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/probe_layout.h" "#define QUAD_PROBE_LAYOUT_ID \"${OOT_NATIVE_LAYOUT_ID}\"\n")
add_library(quad_probe SHARED provider.c)
set_target_properties(quad_probe PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
target_include_directories(quad_probe PRIVATE ${OOT_NATIVE_INCLUDE_DIRS} "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(quad_probe PRIVATE ${OOT_NATIVE_COMPILE_DEFINITIONS})
if(NOT MSVC)
    target_link_libraries(quad_probe PRIVATE m)
endif()
foreach(config Debug Release RelWithDebInfo MinSizeRel)
    string(TOUPPER "${config}" upper)
    set_target_properties(quad_probe PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${upper} "${CMAKE_BINARY_DIR}/mod/provider")
endforeach()
configure_file(main.lua "${CMAKE_BINARY_DIR}/mod/main.lua" COPYONLY)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS manifest.toml)
file(READ manifest.toml manifest_text)
file(WRITE "${CMAKE_BINARY_DIR}/mod/manifest.toml" "${manifest_text}host_layout_id = \"${OOT_NATIVE_LAYOUT_ID}\"\n")
